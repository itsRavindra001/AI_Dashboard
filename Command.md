# AI Dashboard — Command Cheat Sheet

## Conda

# Activate project environment
conda activate D:\Development\ai_ml_env

# Check Python version
python --version

# Check which Python is being used
where.exe python

# Deactivate environment
conda deactivate


## PowerShell

# Show current folder
pwd

# List files
ls

# Enter a folder
cd folder_name

# Go back
cd ..

# Clear terminal
cls


## Git

# Check Git version
git --version

# Initialize Git repository
git init

# Check repository status
git status

# Stage files
git add .

# Create a commit
git commit -m "commit message"

# Check Git branches
git branch

# Rename branch to main
git branch -M main

# Add GitHub repository
git remote add origin <github-repository-url>

# Check remote repository
git remote -v

# Push code to GitHub
git push -u origin main

# Push future changes
git push
